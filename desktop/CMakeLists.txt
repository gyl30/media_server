cmake_minimum_required(VERSION 3.22)
project(gb_monitor VERSION 0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network WebSockets)
find_package(Boost 1.89 REQUIRED CONFIG)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED libavformat>=60 libavcodec>=60 libavutil>=58 libswscale>=7 libswresample>=4)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.20)

add_library(desktop_media STATIC decode_thread.cpp finite_http_input.cpp playback_view.cpp platform_client.cpp record_query_dialog.cpp record_thread.cpp history_request.cpp talk_session.cpp)
target_include_directories(desktop_media PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/.." ${FFMPEG_INCLUDE_DIRS})
target_link_directories(desktop_media PUBLIC ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(desktop_media PUBLIC Qt6::Widgets Qt6::Network Qt6::WebSockets PkgConfig::SDL2 ${FFMPEG_LIBRARIES})
target_link_libraries(desktop_media PRIVATE Boost::headers)
target_compile_options(desktop_media PRIVATE -Wall -Wextra -Werror ${FFMPEG_CFLAGS_OTHER})
add_executable(gb_monitor main.cpp monitor_window.cpp)
target_link_libraries(gb_monitor PRIVATE desktop_media)
target_compile_options(gb_monitor PRIVATE -Wall -Wextra -Werror)
set_target_properties(gb_monitor PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
